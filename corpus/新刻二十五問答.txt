《𗹌𘐘𗍫𗰗𗏁𗤄𗊬》

𘈒𗤄：𗢳𘒣：「𗎫𘟀𗢳𗵆」𘘣𗫂𘌽𗧘𘝶𗦜？𘘚𘒣：𗰜𗎫𗮔𗩱𗫂，𘟀𘟂。𗰜𗎫𘟀𗩱，𘟀𘟣𗦇𗤋𗫂，𗮔𘟂。𗮔𗰜𗋕𗡶，𘉐𗭍𗅋𗋃𗫂，𗎫𘟂。𘌽𗍊𗎫𘟀𗗙𗢳𗵆𘘣𘃞。𗅉𗤄：𘎳𗫻𘁟𗈞𗥃𘍦𗧘𗫂，𘓂𘙌𘟂？𘘚𘒣：𗈪𗆫𘟀𗩱，𗌭𘎳𗧘𘟂；𘟀𘘂𘟀𘙇𗫡𗢳𗵆𗫂，𗫻𗧘𘟂。𘓷𘟀𗎫𘟂。𗎫𗫂𗵘𘟂；𗵘𗫂𘉐𘟂；𘉐𗫂𗢳𘟂。𘌽𗫂𘁟𗧘𘟂。𗎫𘟀𗰜𘖀𗲠𗘺，𘟀𗦇𗅋𘟣，𘟀𘝵𘟀𗤋，𘘦𗈞𗧘𘟂。𗤄𗇋𘌽𗓨𗪲𗒐𘒩𘕤。

𘈒𗤄：𘘝𘈷𗤶𗭍𘅍，𗏹𗮔𗅋𘜘。𘂳𘂳𗢐𗍣，𘓂𘙌𗧘𗦇？𘘚𘒣：𗍣𗢐𗈪𗄻𘉞？𘈷𘒣：𗄻𗧓。𘘚𘒣：𗄻𗌭𗋕𘟂。𗄻𗫻𗉣𘃡𘅇𗣬。𗄻𗫻𗉣𘃡𗌭，𗋕𗄻𘝵𘗫𗨻。𗰜𗎫𗑗𘟀，𗌭𗒘𗵘𗗙𘆊𘟂。𗰜𗎫𗄻𘟀，𗌭𗋕𘝵𗊮𗍣𗶷。𗎫𘊛𗋽𗍊𘜑𗄈，𘕣𘉰𗋽𗗙𗊋？𗎫𗰜𘕿𗅋𗚛𗍣，𘖑𗄻𗌭𗫦𗷝𗫡𘟂。𗤄𗇋𗆰𗍣𘚻𘅎𗞞𗨛。

𘑬𗤄：《𘃣𘉒𗖰》𘒣：「𘛥𗌻𗤶𗫻𗌭，𗙏𗓁𗹙𘟂。𘛥𗌻𗤶𗅋𗫻𗌭，𗭩𘓐𗹙𘟂。𘛥𗌻𗅋𘛥𗌻𗤶𗫻𗌭，𘋢𗥤𗹙𘟂」。𗱕𗢳𗹙𗫂𘓂𘙌𘟂𘘣？𘘚𘒣：𘌽𘕕𘛥𗌻𘟀𗩱𗫂，𘗫𗅢𗖵𗄈𘟂。𗤶𘓷𗰜𘕿𘗫𗤋𘝵𘟀𗫂𗌭，𗢳𗹙𘟂。𗤄𗇋𘒣：𗩾𘂤𗅢𘎪𗤋𘃞𘘣。

𘈒𗤄：𗬩𗅆𘟛𗫂，𘓂𘙌𘟂？𘘚𘒣：𗎫𘟀𗘺𗑗𗫂，𗬩𘟂。𗎫𘟀𗘺𗑗，𘓷𗎘𗛲𗤋，𗯨𗛧𗅋𗶻，𗒘𗒘𗅋𗚛𗫂，𗅆𘟂。𘓷𗎫𗎘𗤋，𘊛𗋡𗓐𗍊，𗅁𗬫𗮀𗭼，𗭍𗯩𗦁𗤋𗫂，𘟛𘟂。𗤄𗇋𘒣：𗡞𗕑𘋢𗋐，𘌽𗥤𘈩𗤋。𗟷𗖏𘓇𘍐，𘎪𘏚𗰓𗁁。

𘈒𗤄：𘝦𘝶𗦜𗹢𗌭𗢳𗵆𘜘𘃞？𘘚𘒣：𗆫𗤋𘓷𘟀𗌭，𘝵𗢳𗵆𘟂。𗅉𗤄：𘕣𗍊𗆫𗤋𘃞？𘘚𘒣：𗢳𗨻𗆫𗤋。𗅉𗤄：𗢳𗅋𗨻𗆫𗫂，𘓂𘙌𘟂？𘘚𘒣：𗢳𘎪𘓊𗰣𗹙𗆫𘜘𗦇𗤋𗌭，𗠝𗷧𘏞𘛛𘟂𘘣。𗤄𗇋𘚻𘅎𗞞𗨛。

𘈒𗤄：𗖰𗚩𘂤𘎪：𘛽𗛆𗑝𘀩，𗊴𘎼𗿇𘑅𗳒𗴿𗒀𘃡，𗌭𗢳𗵆𗈪𘜘。𘘚𘒣：𗅋𘜘。𘕣𗖵𗅋𘜘？𘘚𘒣：𗢳𗵆𗫂𗤶𘟂。𗑝𘀩𗫂𘛽𘟂。𘛽𗫂𗼻𗋽𗜐𘚶𘟂，𗢳𗵆𘝶𘜘？𗅉𗤄：𘌽𘒣𗅔𗌭，𘝶𗦜𘜘𘃞？𘘚𘒣：𗎫𘟀𘝵𘜘？𗅉𗤄：𗎫𘕣𘔮𗑠𗴭？𘘚𘒣：𗴭𘔮𗤋𘃞。𗅉𗤄：𘓂𘙌𗧘𗦇？𘘚𘒣：𘟀𗌭𘜤𘟀，𗅋𘟀𗌭，𗅉𗅢𗤽𗳒𘜘𘏚𗅋𗁁。

𘈓𗤄：𘕕𘜶𗠝𗗆𗒯𗑱𗫂，𘓂𘙌𘟂？𘘚𘒣：𗅴𗕣𘚠𘟂。𗅉𗤄：𘟓𗦜𗟥𗧤？𘘚𘕫：𗧤𗫂𘗫𘟂。𗮅𗑱𘕕𗀀𗰜𘕿𘝵𗲠。𘌽𗧘𗄻𗌭𗋕𗡶𘝵𗍣。𗋕𗧤𗥤𗇋，𗧀𗹦𘎳𘜘。𗢳𗵆𗧠𗌭𗄻𗖵𗤶𗭍。𗢳𘎪：「𗉛𗷫𗅋𗧤𗳒，𘈬𗦺𘕿𘃽𘘣」。𗱕𗫦𗇋𘆄，𗎫𗰜𘎳𗤋𗷝𘟀，𗤶𗄈𗳒𗉛𗷫𗧤𗧠，𗧤𗤶𗄈𗫂，𘝵𗉛𗷫𗤶𘟂𘖑𗄻。𘌽𘔼𗇖𗨻，𗧐𗇘𗅋𘜘。𗫈𗹢𗇋𘆄，𗧀𗤶𗅋𗄈，𗆫𗚛𗤋𗌭，𗋕𗡶𗧐𗇘𘃞。

𘈒𗤄：𘋢𗋐𗢳𗵆𗈪𘜘？𘘚𘒣：𗅋𘜘。𗅉𗤄：𘋢𗋐𗅋𘜘𗌭，𗢳𗵆𗇋𗖶？𘘚𘒣：𘋢𗋐𘗫𘟂，𗥃𘍦𗫻，𘂆𗢳𗵆𘜘𗗂？𗢳𗵆𗇋𗫂，𘋢𗋐𗗙𗰜𗎫𘟂。𗅉𗤄：𘋢𗋐𗰜𗎫，𘕣𘔮𗑠𗴭？𘃞？𘘚𘒣：𗰜𗎫𗴭𘔮𗤋，𗯨𘂤𘊛𘏼𗈐，𗴭𗦢𘏚𗸹𗁁𘘣。

𘈒𘑘𗢨𗤄：𘘚𗵘𗈪𗫦？𘘚𘒣：𘀍𗵘𘝶𗦜？𘑘𘒣：𗵘𗫂𗲠𗘺𗗙𗙑𘟂。𗷚𗡅𗩜𗆑，𘄎𗡊𘝣𘎼，𗥕𗹬𗋿？□𘟂。𘘚𘒣：𗵘𘖑𗚂𘉞𘃞。𗧓𗫈𗵘𘎪，𗍳𗋚𗨉𘉞。𗋕𗡶𗵘𗫂，𘋢𗋐𗄑𗄑𗗙𗰜𗎫𘟂。𗰜𗎫𘟀𗌭𘝵𗦳𗘺𗴴。𗋕𗵘𗓰𗤓，𗦾𘅍𗅋𘟀，𗨉𘅍𗅋𗓁𗡢𘅍𗅋𘜘。𗱕𘓐𗾞𗾞𗋕𗭍，𗑷𗄊𘖑𗄻，𗓰𘂤𗩾𗓰𗫔𗤓𘗠𘟂。𘜘𗇋𗏹𗫻，𗹬𗇋𗅋𗇖，𗩱𗇋𗏹𗴴。𘌽𗧘𘔼𗌭𘘦𗵘𘘣𘃞。𘑘𘒣：𗇁𘘚𘕣𘈃𘍳𘉞𘃞𘘣。

𘈒𗤄：𘘚𗍳𗑗𗈪𗸸𘉞？𘘚𘒣：𗑗𗸸𘗫𗳒□。𗅉𗤄：𗢳𘝶𗦜𗵆？𘘚𘒣：𘝵𘓷𗰜𗑗，𗑗𗸸𗰓𗋂？𗓱𗑗𗸸𗌭，𗤶𗈪𗄈𘟂。𗤶𗄈𗇖𗨻。𗇖𗌭𗇔𘃞。𗤶𗄈𗦇𗤋，𗰜𗑗𗄻𗌭，𘘦𗢳𗵆𘟂。

𘈒𗤄：𘉋𗧐𗇘𗫂，𘓂𘙌𘟂？𘘚𘒣：𘉋𗹬𗫂，𘉋𗧐𗇘𗤶𘟂。𗅉𗤄：𗢳𘉋𗹬𗈪𗿷？𘘚𘒣：𗢳𗿷。𗅉𗤄：𘋢𗋐𗈪𗿷？𘘚𘒣：𗿷。𗅉𗤄：𘉋𗹬𗿷𘓁𗈪𗅲，𘕣𗖵𗢳𗧐𗇘𘜘，𘋢𗋐𗅋𗧐𗇘𘃞？𘘚𘒣：𘋢𗋐𘃺𗖵𗛧𘚢，𗢳𘃺𗖵𗅋𘚢，𘔼𘘦𘓷𗧘𘌽𗍊𘃞。

𘈒𗤄：𘝶𗦜𘕕𗐯𗣈𘜘？𘘚𘒣：𗤶𘕕𘗽𗑠𗅋𘌥𗩍𘟀，𗌭𘕕𗐯𗣈𘟂。𗋚𗦎𗅋𗆫，𗷝𗆡𗆫𗤋，𗌮𗫻𗆫𗰛，𘕕𗐯𗣈𘟂。

𘈒𗤄：𘋢𗋐𗢳𘟂，𗢳𘋢𗋐𘟂𘘣𗫂，𘌽𗧘𘝶𗦜？𘘚𘒣：𗎫𘟀𗢳𘟂，𗅋𘟀𘋢𘟂。𗋕𗧘𘌽𗍊。

𘈒𗤄：𗅆𗖵𘟛𘃽𗌭𗈪𘟂？𘘚𘒣：𗅋𘃞。𘟛𗖵𗅆𘃽𗌭𗈪𘟂？𘘚𘒣：𗅋𘃞。𘌽𗫂𘝶𗦜？𘘚𘒣：𗅆𗖵𘟛𘃽𗌭𗙏𗓁𗹙𘟂。𘟛𗖵𗅆𘃽𗌭𘞪𗫨𗹙𘟂。𗅆𘟛𗣼𗳦𗌭，𘋢𗥤𗹙𘟂。𗣼𗅆𗅆𗅋𘟣，𗣼𘟛𘟛𘍦𗤋，𘌽𗫂𗢳𗹙𘟂𘃞。

𘈒𗤄：𘓷𗫂𘓂𘙌𘟂？𗎫𗫂𘝶𗦜𘃞？𘈩𘟂𗗂，𘁟𘟂？𘘚𘒣：𘓷𗖵𘈩𘟂，𗎫𗖵𘁟𘃞。𘌽𗫂𘝶𗦜？𘘚𘒣：𘓷𘊛𘗊𗍊，𗎫𘊛𗪼𗍊，𘌽𘔼𘁟𘃞。𗅉𗤄：𗪼𘗊𗳒𘃡，𘕣𗖵𘁟𘃞？𘘚𘒣：𗪼𗫂𘗊𘓁𘟂，𗫶，𘔮𗮔𘖑𗩱。𘃶𗪼𗨻𘅍，𘘦𘔮𗮔𗩱𘃞。𗋕𗖵𘁟𘃞。𘋢𗋐𗄑𗄑𗄊𗢳𗎫𗫔，𘜶𗩴𗄻𗹬𗅆𘈨𗖢𗋂，𘘦𗰜𗤶𘟀。𗰜𗤶𘟀𗅉，𘝵𗸐𗹢𗭍，𘘦𗮔𗑗𘜘𘟂。𗅔𘟂𗏴𗩯，𗠷𗷎𗤶𗤋，𘓊𗽀𗅋𗨙，𗕑𘍦𗘯𗜓，𘂆𗣛𘙇𗆐𗅔；𗪼𘓷𗏹𗮔，𘂆𗋕𘙇𗶹𗅔。𗕑𘍦𘛒𘛒，𗪼𘓷𗘺𗘺，𗶷𗄼𗅋𘟣，𗣛𗳺𘂆𗤋。𗹢𗇋𗤶𗭍，𘌽𗖵𘝇𗦇，𗱕𗢳𗎫𘂆𘌽𗑠𗈪𗅲。

𘈒𗤄：𘘝𘈷𗌭𗤶𗭍𘅍，𗋕𗡶𗲠𗘺𗨻𗫂𗈪𘟂？𘘚𘒣：𗲠𗘺𘟀𗌭，𗧀𗉣𘃡𘟂。𗅉𗤄：𘓂𘙌𗧘𗦇？𘘚𘒣：𗹢𗇋，𗍳𗌮𗤶𗭍𗿳，𘟀𘟀𗦇𗤋。𗲠𗘺𘝶𘟀？𗢳𘒣：𘜘𗦇𘓊𗰣𗹙𗤋，𗌭𗠝𗷧𘏞𘛛𘟂𘘣。

𘈒𗤄：𘗫𗄈𗫨𗌭𗧐𗇘𗈪𘜘？𘘚𘒣：𗅋𘜘。𗅉𗤄：𘗫𗈞𗫨𗈞𗌭𗈪𘜘？𘘚𘒣：𗅋𘜘。𘝶𗦜𘜘𘃞？𘘚𘒣：𘗫𗄻𗖵𘗫𗨻，𗫨𗄻𗖵𗫨𗨻。𘝵𗤶𗲠𗘺，𗄻𗅋𗫻𗌭，𗲠𗘺𗢳𗤶𗌭𘟂。𗋕𗧘𘂆𘌽𗍊𘃞。

𘈒𗤄：𘋢𗋐𘍒𘃡𗌭，𘘍𗷩𗈪𗇔？𘘚𘒣：𗇔。𗅉𗤄：𗢳𗎫𗈪𗇔？𘘚𘒣：𗇔。𗅉𗤄：𘋢𗋐𗇔𗌭，𘍔𗨙；𗢳𗎫𗇔𗫂，𘍔𗨙𗅋𗨙？𘘚𘒣：𗢳𗎫𘍔𗅋𗨙。𗅉𗤄：𘕣𘃞？𗨙𘃞𗅋𗨙𘃞？𘘚𘒣：𘊛𗍊𗵒𘛇𗜐𘂅𗅁𘋃，𘛇𘍦𘐠𗹪，𗵒𗎫𗋕𗡶𗰜𘕿𗅋𗈭。𗅋𗈭𗌭𗰜𗅆𘟂。𗰜𗅆𗌭𗏹𗦻𗨻。𗒘𗏹𗌭𘐠𗹪𘂆𗤋。𗨙𗦇𗰓𘟣𘃞？

𘈒𗤄：𗅆𘟛𗣼𗳦𗧘𗫂𘓂𘙌𘟂？𘘚𘒣：𗎫𗅋𗚛𗌭𗅆𘟂；𗅋𗚛𘟀𗩱𗫂，𘟛𘟂。𘝦𗭍𗋕𗡶𘅙𗤋𗌭𗣼𗳦𘟂。𘌽𗖵𗰜𗎫𗮔𗩱，𗌭𗢳𗎫𘟀𘟂。

𘈒𗤄：𘍞𗢳𘝶𗦜𗵆？𘘚𘒣：𗢳𗅉𘋢𗋪，𘃞𗿳𘅇𗆫，𗋕𘏚𗥲𗇘。𗤄：𘝶𗦜𗖵𗡶𘜘？𘘚𘒣：𗩴𘊄𗄑𗄑𗄊𗅋𗾫，𗌭𘝵𗢳𗎫𘟀，𗖵𗡶𘜘𘃞。

𘈒𗤄：𗰜𘕿𗫌𘐩，𗢳𗵆𗑩𘔼𘟂。𘝶𗦜𗤶𗭍𘂎𗢳𗵆𘜘𘃞？𘘚𘒣：𗭍𗦇𗤶𗤋，𘝵𗢳𗵆𘜘。𗤄：𗓱𗤶𗤋，𗌭𗢳𗖶𗨻𘃞？𘘚𘒣：𗤶𗤋𘝵𗢳𗵆。𗢳𗵆𘂆𗤶𗤋。𗤄：𗢳𘜶𗕿𘜶𗈁𗅢𘎪𘏚𗤋𗪺，𗿷，𘋢𗋐𗌗𗩱。𗓱𗤶𗤋，𗌭𘋢𗋐𗖶𗌗？𘘚𘒣：𗤶𗤋𘜘，𗌭𘋢𗋐𗌗𗒘𘟂。𗓱𗌗𗦇𘋢𗋐𘟀，𗌭𗧀𗤶𘟣𗨻。𗤶𗓱𘟣，𗌭𗌮𗢠𘎳𘟂。𗤄：𗤶𗓱𗤋𗌭𗍣𘟀𗋸𗨻？𘘚𘒣：𘗫𗤶𗰜𗤋，𗩂𗄻𗅋𗍣，𗹏𘟀𘝶𗨻？

𘈒𗤄：𗈪𗆫𗖵𗡶𗫂𘓂𘙌𘟂？𘘚𘒣：𘃺𘄡𗍱𗤋，𘝵𗖵𗡶𗨻。𗤄：𘃺𘄡𗍱𗤋𗌭，𗢳𗎫𗖶𘟀？𘘚𘒣：𘃺𘄡𗤋，𗌭𘓷𗮔𘞪𘋮，𘝵𗳒𘝵𗅋𘟀𘃞。

𘈒𗤄：𘋢𗋐𗢳𗎫𗑠𘁟𗂙𗤋，𗌭𘈩𘓐𘝦𗹢𗢳𗨻𘅍，𘋢𗋐𗄑𗄑，𘃞𗿳𗳦𗳦𗄊𗧐𗇏𗟭。𗫈𘌽𗅋𗍊𗫂，𗋕𗧘𘝶𗦜？𘘚𘒣：𗍳𗤻𗡮𘂤𗤁𘍦𗧘𘖑𗉘𗐵𘉞𗗂？𗯮𘂤𘁟𘟣，𘁟𘂤𗯮𘟣，𗵆𘂤𗹪𘟣，𗹪𘂤𗵆𘟣，𗦬𘂤𗂙𘟣，𗂙𘂤𗦬𘟣。𘋢𗋐𗢳𗑠𘈩𗎫𗯮𗫶，𗯩𘎆𗅋𗦁，𗪺𗤁𘖑𗳦，𘝵𗹢𘝵𘜘。𘉍𗔉𗠰𗦾，𘝵𘘂𗅋𘀞。𘊛𗿼𗴮𘆄𗄑𗄑𘈩𗲠𗐯𗦬，𗤁𘖑𗳦𗖵，𘕥𗦇𗲠𘁟。𘝋𘝄𗤋𗫂，𗲠𗐯𗫻𗫶，𗼻𗑠𗅋𗈜。𘝄𘙌𗄭𘂆𗤢𗹭𘟣𘔼，𗲠𘕥𗡶𘋻。𗿻𘚚𗿼𗤁，𗲠𗎘𗫡，𗉘𗅉𗿼𘜔，𗋕𗑠𘜄𗟻𘏚𗰓𗁁？𘘦𘋢𗋐𗄑𗄑，𘈩𗢳𗎫𗦬，𘄡𗤁𗌭𘖑𗳦，𘂆𘌽𗑠𗈪𗅲。

𘈒𗤄：𗇁𗹢𗤶𘕋𗈜𗡶𘝶𗦜𘟂？𘘚𘒣：𗍳𘛽𗤶𗅁𗶂𗶂𗋚𘝯𘉞？𗏁𘜭，𗰗𗍫𘋩，𗰗𘉋𗐯𘃞𘃞，𗺉𗋒𗌭𗽀𗰣𘔮，𘜘𘏚𗈪𗁁？𗊬𘒣：𗫈𘛽𗤶𗅁，𗶂𗶂𗋚𘝯，𗌭𘈩𘔮𘜘𗦇𘟣𘖑𗉘(𘟀)𗧓。𘘚𘒣：𗍳𘛽𗤶𘍦𗹫𗞞𗨻？𗊬𘒣：𘛽𗤶𘝵𘍦𗈜，𗹫𗦇𘕣𘟣？𘘚𘒣：𗍳𘛽𗤶𘃵𗅉，𗂙𘔮𗈪𘟣？𗊬𘒣：𘛽𗤶𘝵𗤋𘂤，𗅉𗬫𘔮𗰓𘟣？𘘚𘒣：𗍳𗯨𘂤𘍦𗹫𗞞𗨻？𗊬𘒣：𗯨𘂤𘍦𘝵𘍦𗤋，𗹫𗰓𗋂？𘘚𘒣：𘌽𗍊𗌭𗍳𗫈𘕋𗈜𘟂。𗤄𗇋𘌽𗖵𗮔𗥤，𗡙𘒣：𗅢𘎪𘏚𗤋，𘒩𘘝𗋚𗨙。

𘈒𗤄：𘌽𗍊𗥤𘟀，𗅉𗏣𗎭𘕧𗳒𘝦𗹢，𗋂𗗂？𘘚𘒣：𗎭𘕧𗳒𗹢𘂆𘃜𘟣，𗎭𗅋𘕧𗳒𗹢𘂆𘃜𘟣。《𘉂𘀲𗖰𗚩》𘒣：「𘈩𗼻𘕿𗄈，𘈩𗼻𘕿𗫡𗳒𗅔，𗘺𗈞𗌮𗒘𗎭𘐡𗰓𘟣？」《𗾫𘞙𗖰》𘒣：「𗓱𘓐𗣼𗎫𗓁𗌭，𘈩𗼻𘕿𗄈，𘈩𗼻𘕿𗫡𗅋𘟂。𘌽𘓐𗢣𘎳𗅋𘘢，𘈬𗦺𗅋𗫻𘘣。」𗅉《𗺀𘄾𗖰𗖍》𘒣：𗪘𘃽，𘈩𗄼，𗅋𗆮，𗠝𗓽𘂦，𘌽𗍊𗱕𗹦𗼃𘓐，𗄊𘗫𗤶𗖵𘟣。」𗓑𗨁𗡪𗧘𗆤𗖵𘟂，𗌭𘞌𘍦𗗙𗧘，𘄎𗑗𗫨，𗤶𘞌𘝦𗎭𘜘𗦇𗤋。𗓱𗯨𗪙𗼄𘕤𘕣𗖵，𗌭𘝦𗎭𘕧𘂆𗦁𗄿𗅋𘟣。𘍞𘅣𗳒𗧘𗅋𗦁，𗧘𗳒𘅣𗅋𗦁，𗧘𘅣𗦁𗤋𗌭𗾞𗾞𗭍𗳒𗭍𗤋𗑠𗅋𘇄，𗅋𗹢𘏚𗰓𗁁？𗓱𘅣𗂆𗧘𗈦，𗧘𗂆𘅣𗈦，𗌭𗖵𗡶𘝶𘜘？𗋕𘒣𗅔𘃞。𗤄：𘌽𗹙𘗠𗫂𗦻𘍦𗈪𘟣？𘘚𘒣：𘌽𗹙𗰜𘕿𗦻𘍦𗅋𘟣。𗦻𘍦𗤋𘂤𗏗𗦻𘍦𘊴。𘌽𗹙𘗠𗦻𗫂「𗵒𘗁𘋅𗁲」𘘣，𗹪𘏚𗤋𘃞。𗅉𗦻「𗱭𗺀𗶮𘋅𗁲」𘘣，𗠁𗰛𗤋𘃞。𗅉𗦻「𗹙𗎫𘋅𗁲」𘘣，𗓆𗈭𗤋𘃞。𗅉𗦻「𗧐𗇘𘋅𗁲」𘘣，𗆂𗇖𗤋𘃞。𘌽𗧘𗥤𗩱𗌭，「𗵒𘗁𘜶𗇋」𘘣，𗢐𗩱𗇋𗤋𘃞。𗤄：「𗵒𘗁𘜶𗇋」𗣀𘉐𘕣𗿷？𘘚𘒣：𘌽𗗙𗣼𘉐𗧀𗢳𗄻𗩱，𗈪𗆫𗖵𗡶𗿳，𗣋𘔏𗉛𗷫𗈞𗩱，𗃣𗦇𗅋𘟣。𗣋𘔏𗣼𘉐𗄭𗩱，𘖑𗍿𗅋𘟣。𗱕𗹦𘉋𗴮𗵃𗼈𘆄𗗙𗏹𗶅𗓱𗦇𗨻。𗶷𗯩𗦁𗤋，𘆅𗹛𘟙𗍊。𘄡𘟛𘉍𗭺，𘜶𗾔𗋚𗍊。𗅉𘎳𘅇𗪆，𗰜𘝵𘎳𗤋，𗅉𗈞𘅇𗪆，𗰜𘝵𗈞𗤋。《𗤻𗡮𗖰》𘒣：「𗹙𗄑𗄑𗅋𘎳，𗹙𗄑𗄑𗅋𗈞。𗓱𘌽𗥤𗩱𗌭，𗱕𗢳𗌮𗙼𗜓」𘘣。𘌽「𗑛𗰜𗤶𗼻𗹙𘗠」𘃵𗅉，𗂙𘈩𘓊𗽀𗰣𗹙𘟣𘘣，𗌭𗋕𗫂𗮀𗤋𘈟𗏅𗷝𗟥𗇋𘟂。𗅉𘋠𘋠𗹙𘟣𘘣，𗤶𘃵𗅉𘓆𘞞𗇋𗫂，𘜶𗈦𘟂。𘕿𘘣𗦇𘕣𘂬？𗋕𗍊𘓐𗑠𗟲𗅋𗍫𗦇。𗱕𗹢𗇋𘆄𘋦𗦇𘋦𗦇。

《𗁬𗕸𗂧𘘚𘉍𗃵𗫔𗎭𗅁𗫻𗉘𗿳𗫔𘓐𗢳𗧘𗞞𗤄𗍫𗰗𗏁𗤄𗊬》𘃪。
